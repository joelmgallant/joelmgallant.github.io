# Music Web Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `/music/` on joelmgallant.com: a stable, pannable canvas map of the artists Joel listens to, clustered by sound, with a lineage drill-down and a punk-to-folk story.

**Architecture:** A local Node pipeline in `tools/music-web/` reads the Apple Music library (JXA), enriches artists from MusicBrainz and Last.fm with disk caching, computes clusters (Louvain) and positions (seeded d3-force) offline, and writes `static/music/data.json`. The page in `static/music/` is vanilla ES modules drawing on canvas 2D, with no framework and no build step. Hugo serves `static/` verbatim; pushing `trunk` deploys.

**Tech Stack:** Node 24 (`node:test`), `osascript -l JavaScript`, `d3-force@3.0.0`, `graphology@0.26.0`, `graphology-communities-louvain@2.0.2`, vanilla JS + canvas 2D, headless Chrome for visual checks.

**Spec:** `docs/superpowers/specs/2026-11-17-music-web-design.md`

- MusicBrainz: at most 1 request per second, `User-Agent: music-web/0.1 ( https://joelmgallant.com/music/ )`. No personal email anywhere.
- `tools/music-web/cache/` and `tools/music-web/node_modules/` are gitignored. The raw library dump is never committed.
- Published per artist: name, cluster, position, love, songs, plays, favourites, Apple genres, track titles with play counts, similar artists, lineage. **Not published:** dates added.
- Selection rule defaults: 3+ songs, or 10+ plays, or any favourite. Love score: `songs * 1 + plays * 0.2 + favourites * 3` (weights in `overrides.json`). Dot radius: `2 + 1.6 * sqrt(love)`.
- Credits like "A & B" split into separate artists **only when every part exists as an artist in the library on its own** ("Mumford & Sons" stays whole).
- Low-confidence MusicBrainz matches stay unmatched and are listed in the report. Never guess.
- Copy on the page is in Joel's voice: dry, short, plain, honest parentheticals. No punchline on every element, no "(made up)" style hedges, no invented personal stories.
- Front-end modules stay under 200 lines each. Pure helpers must not touch the DOM at import time so `node:test` can import them.
- All motion is disabled under `prefers-reduced-motion: reduce`.
- Publishing (from Task 10 on): push `trunk`, wait for the "Deploy Hugo site to Pages" run for that commit to complete with success, then `curl` the live URL and check for the expected content.

function stripDiacritics(text) {
  return String(text).normalize("NFKD").replace(/[̀-ͯ]/g, "");
}

export function normalizeName(name) {
  return stripDiacritics(name)
    .toLowerCase()
    .replace(/&/g, " and ")
    .replace(/[^a-z0-9]+/g, " ")
    .trim()
    .replace(/^the /, "")
    .replace(/\s+/g, " ");
}

export function slugify(name) {
  const slug = stripDiacritics(name)
    .toLowerCase()
    .replace(/&/g, " and ")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
  return slug || "artist";
}

const CREDIT_SEPARATOR = /\s*(?:,|&|\+|\bfeat\.?|\bft\.?|\bx\b|\band\b|\bwith\b)\s*/i;

export function splitCredit(credit, knownNames) {
  const parts = credit.split(CREDIT_SEPARATOR).map((part) => part.trim()).filter(Boolean);
  if (parts.length < 2) return [credit];
  const everyPartKnown = parts.every((part) => knownNames.has(normalizeName(part)));
  return everyPartKnown ? parts : [credit];
}
```

import { execFileSync } from "node:child_process";

export const JXA_SCRIPT = `
const tracks = Application("Music").libraryPlaylists[0].tracks;
JSON.stringify({
  title: tracks.name(),
  artist: tracks.artist(),
  album: tracks.album(),
  genre: tracks.genre(),
  year: tracks.year(),
  plays: tracks.playedCount(),
  favourite: tracks.favorited()
});`;

export function rowsFromColumns(columns) {
  const count = columns.title.length;
  for (const [key, values] of Object.entries(columns)) {
    if (values.length !== count) throw new Error(`column ${key} has ${values.length} values, expected ${count}`);
  }
  const rows = [];
  for (let index = 0; index < count; index++) {
    rows.push({
      title: columns.title[index] || "",
      artist: (columns.artist[index] || "").trim(),
      album: columns.album[index] || "",
      genre: columns.genre[index] || "",
      year: columns.year[index] || null,
      plays: columns.plays[index] || 0,
      favourite: columns.favourite[index] === true,
    });
  }
  return rows;
}

export function exportLibrary() {
  const output = execFileSync("osascript", ["-l", "JavaScript", "-e", JXA_SCRIPT], {
    encoding: "utf8",
    maxBuffer: 64 * 1024 * 1024,
  });
  const rows = rowsFromColumns(JSON.parse(output));
  if (rows.length === 0) throw new Error("Music library export returned 0 tracks");
  return rows;
}
```

`tools/music-web/select.mjs`:
```js
import { normalizeName, slugify, splitCredit } from "./lib/names.mjs";

export const DEFAULT_RULE = { minSongs: 3, minPlays: 10, anyFavourite: true };
export const DEFAULT_WEIGHTS = { songs: 1, plays: 0.2, favourites: 3 };

function byFrequency(counts) {
  return [...counts.entries()]
    .sort((left, right) => right[1] - left[1] || left[0].localeCompare(right[0]))
    .map(([value]) => value);
}

function increment(counts, value) {
  counts.set(value, (counts.get(value) || 0) + 1);
}

export function groupByArtist(tracks) {
  const knownNames = new Set(tracks.filter((track) => track.artist).map((track) => normalizeName(track.artist)));
  const groups = new Map();
  for (const track of tracks) {
    if (!track.artist) continue;
    for (const part of splitCredit(track.artist, knownNames)) {
      const key = normalizeName(part);
      if (!key) continue;
      if (!groups.has(key)) {
        groups.set(key, { key, spellings: new Map(), genres: new Map(), songs: 0, plays: 0, favourites: 0, tracks: [] });
      }
      const group = groups.get(key);
      increment(group.spellings, part);
      if (track.genre) increment(group.genres, track.genre);
      group.songs += 1;
      group.plays += track.plays;
      if (track.favourite) group.favourites += 1;
      group.tracks.push({ title: track.title, plays: track.plays });
    }
  }
  return [...groups.values()].map((group) => ({
    key: group.key,
    name: byFrequency(group.spellings)[0],
    songs: group.songs,
    plays: group.plays,
    favourites: group.favourites,
    appleGenres: byFrequency(group.genres),
    tracks: group.tracks.sort((left, right) => right.plays - left.plays || left.title.localeCompare(right.title)),
  }));
}

export function loveScore(group, weights = DEFAULT_WEIGHTS) {
  return group.songs * weights.songs + group.plays * weights.plays + group.favourites * weights.favourites;
}

export function passesRule(group, rule = DEFAULT_RULE) {
  return group.songs >= rule.minSongs || group.plays >= rule.minPlays || (rule.anyFavourite && group.favourites > 0);
}

export function selectArtists(groups, { rule = DEFAULT_RULE, weights = DEFAULT_WEIGHTS, forceIn = [], forceOut = [] } = {}) {
  const forcedIn = new Set(forceIn.map(normalizeName));
  const forcedOut = new Set(forceOut.map(normalizeName));
  const chosen = groups
    .filter((group) => !forcedOut.has(group.key) && (forcedIn.has(group.key) || passesRule(group, rule)))
    .sort((left, right) => left.key.localeCompare(right.key));
  const usedIds = new Set();
  return chosen.map((group) => {
    const base = slugify(group.name);
    let id = base;
    for (let suffix = 2; usedIds.has(id); suffix++) id = `${base}-${suffix}`;
    usedIds.add(id);
    return { ...group, id, love: Math.round(loveScore(group, weights) * 10) / 10 };
  });
}

export function missingForcedIn(groups, forceIn) {
  const keys = new Set(groups.map((group) => group.key));
  return forceIn.filter((name) => !keys.has(normalizeName(name)));
}
```

**Interfaces:**

  - `createFetcher({ cacheDir, minIntervalMs, userAgent, fetchImpl?, sleep?, now?, maxRetries? }): { getJson(url, { cacheKey? }): Promise<any>, stats: { network: number, cached: number } }`. Cache file name is `sha256(cacheKey).json`; `cacheKey` defaults to `url`. Retries 429/503 with `1000 * 2^attempt` ms backoff up to `maxRetries` (default 4). ```js
import { createHash } from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";

export function redact(url) {
  return String(url).replace(/api_key=[^&]+/g, "api_key=REDACTED");
}

const realSleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export function createFetcher({
  cacheDir, minIntervalMs, userAgent,
  fetchImpl = globalThis.fetch, sleep = realSleep, now = () => Date.now(), maxRetries = 4,
}) {
  let lastRequestAt = -Infinity;
  const stats = { network: 0, cached: 0 };

  async function readCache(file) {
    try {
      return { hit: true, data: JSON.parse(await readFile(file, "utf8")) };
    } catch (error) {
      if (error.code === "ENOENT") return { hit: false };
      throw error;
    }
  }

  async function getJson(url, { cacheKey = url } = {}) {
    const file = join(cacheDir, `${createHash("sha256").update(cacheKey).digest("hex")}.json`);
    const cached = await readCache(file);
    if (cached.hit) {
      stats.cached += 1;
      return cached.data;
    }
    for (let attempt = 0; ; attempt++) {
      const wait = lastRequestAt + minIntervalMs - now();
      if (wait > 0) await sleep(wait);
      lastRequestAt = now();
      const response = await fetchImpl(url, { headers: { "User-Agent": userAgent, Accept: "application/json" } });
      stats.network += 1;
      const retryable = response.status === 429 || response.status === 503;
      if (retryable && attempt < maxRetries) {
        await sleep(1000 * 2 ** attempt);
        continue;
      }
      if (!response.ok) throw new Error(`HTTP ${response.status} for ${redact(url)}`);
      const data = await response.json();
      await mkdir(cacheDir, { recursive: true });
      await writeFile(file, JSON.stringify(data));
      return data;
    }
  }

  return { getJson, stats };
}

import { normalizeName } from "./lib/names.mjs";

export const MB_BASE = "https://musicbrainz.org/ws/2";
export const MB_USER_AGENT = "music-web/0.1 ( https://joelmgallant.com/music/ )";
export const MB_MIN_INTERVAL_MS = 1100;

const RELATION_LABELS = {
  "member of band": { forward: "member of", backward: "member" },
  subgroup: { forward: "related group", backward: "related group" },
  "is person": { forward: "same person", backward: "same person" },
};

export function searchUrl(name) {
  const query = `artist:"${name.replace(/"/g, "")}"`;
  return `${MB_BASE}/artist?query=${encodeURIComponent(query)}&fmt=json&limit=5`;
}

export function lookupUrl(mbid) {
  return `${MB_BASE}/artist/${mbid}?inc=artist-rels+genres&fmt=json`;
}

export function pickMatch(searchJson, name, minScore = 95) {
  const wanted = normalizeName(name);
  const candidates = (searchJson.artists || [])
    .filter((candidate) => candidate.score >= minScore && normalizeName(candidate.name) === wanted)
    .sort((left, right) => right.score - left.score);
  if (candidates.length === 0) return { mbid: null, reason: "no confident match" };
  const best = candidates.filter((candidate) => candidate.score === candidates[0].score);
  if (best.length > 1) return { mbid: null, reason: `ambiguous: ${best.length} artists named "${name}"` };
  return { mbid: best[0].id, reason: null };
}

export function parseLineage(artistJson) {
  const seen = new Set();
  const lineage = [];
  for (const relation of artistJson.relations || []) {
    const labels = RELATION_LABELS[relation.type];
    if (!labels || !relation.artist) continue;
    const label = labels[relation.direction] || labels.forward;
    const key = `${label}|${relation.artist.id}`;
    if (seen.has(key)) continue;
    seen.add(key);
    lineage.push({ relation: label, name: relation.artist.name, mbid: relation.artist.id });
  }
  return lineage.sort((left, right) => left.relation.localeCompare(right.relation) || left.name.localeCompare(right.name));
}

export function parseGenres(artistJson) {
  return (artistJson.genres || [])
    .slice()
    .sort((left, right) => right.count - left.count || left.name.localeCompare(right.name))
    .slice(0, 5)
    .map((genre) => genre.name);
}

export async function enrichMusicBrainz(artists, { fetcher, pinned = {} }) {
  const enriched = [];
  const unmatched = [];
  for (const artist of artists) {
    let mbid = pinned[artist.id] || null;
    let reason = null;
    if (!mbid) ({ mbid, reason } = pickMatch(await fetcher.getJson(searchUrl(artist.name)), artist.name));
    if (!mbid) {
      unmatched.push({ id: artist.id, name: artist.name, reason });
      enriched.push({ ...artist, mbid: null, mbGenres: [], lineage: [] });
      continue;
    }
    const details = await fetcher.getJson(lookupUrl(mbid));
    enriched.push({ ...artist, mbid, mbGenres: parseGenres(details), lineage: parseLineage(details) });
  }
  return { artists: enriched, unmatched };
}

import { readFileSync } from "node:fs";
import { homedir } from "node:os";
import { join } from "node:path";

export const LASTFM_BASE = "https://ws.audioscrobbler.com/2.0/";
export const LASTFM_MIN_INTERVAL_MS = 250;
export const LASTFM_USER_AGENT = "music-web/0.1 ( https://joelmgallant.com/music/ )";

const GENERIC_TAGS = new Set([
  "seen live", "favorites", "favourites", "favorite", "favourite", "male vocalists", "female vocalists",
  "american", "british", "canadian", "usa", "uk", "english", "australian", "irish", "scottish", "all",
]);

export function readApiKey(path = join(homedir(), ".secrets/tokens/lastfm-api-key")) {
  const key = readFileSync(path, "utf8").trim();
  if (!/^[0-9a-f]{32}$/.test(key)) throw new Error(`Last.fm key at ${path} is not a 32-character hex string`);
  return key;
}

export function methodUrl(method, artistName, apiKey) {
  const params = new URLSearchParams({ method, artist: artistName, autocorrect: "1", limit: "100", format: "json" });
  const cacheKey = `${LASTFM_BASE}?${params}`;
  params.set("api_key", apiKey);
  return { url: `${LASTFM_BASE}?${params}`, cacheKey };
}

export function parseSimilar(json) {
  if (json.error || !json.similarartists) return [];
  return json.similarartists.artist
    .map((artist) => ({ name: artist.name, match: Number(artist.match) }))
    .filter((artist) => artist.match > 0);
}

export function parseTopTags(json) {
  if (json.error || !json.toptags) return [];
  return json.toptags.tag
    .map((tag) => ({ name: tag.name.toLowerCase().trim(), count: Number(tag.count) }))
    .filter((tag) => tag.count >= 10 && !GENERIC_TAGS.has(tag.name))
    .slice(0, 10);
}

export async function enrichLastfm(artists, { fetcher, apiKey }) {
  const enriched = [];
  const missing = [];
  for (const artist of artists) {
    const similarRequest = methodUrl("artist.getSimilar", artist.name, apiKey);
    const tagsRequest = methodUrl("artist.getTopTags", artist.name, apiKey);
    const lfmSimilar = parseSimilar(await fetcher.getJson(similarRequest.url, { cacheKey: similarRequest.cacheKey }));
    const lfmTags = parseTopTags(await fetcher.getJson(tagsRequest.url, { cacheKey: tagsRequest.cacheKey }));
    if (lfmSimilar.length === 0 && lfmTags.length === 0) missing.push({ id: artist.id, name: artist.name });
    enriched.push({ ...artist, lfmSimilar, lfmTags });
  }
  return { artists: enriched, missing };
}

  - `nameClusters(clusterOf, artists, nameOverrides = {}): { id, name, color, size }[]`: `nameOverrides` maps a **member artist id** to a cluster name. Auto names are the top two tags, title-cased, joined with `" / "`, never `"alternative"`. `misc` is `"Everything else"`.
  - `computePositions(ids, edges, previous = {}, { seed = 42, ticks = 400, anchorStrength = 0.5 }): Record<id, { x, y }>`: deterministic, rounded to 0.1.
  - `runLayout(artists, previous, { clusterNames, resolution, seed }): { edges, clusterOf, clusters, positions }`
  - `MISC_CLUSTER = "misc"`

import Graph from "graphology";
import louvain from "graphology-communities-louvain";
import { forceSimulation, forceLink, forceManyBody, forceCollide, forceX, forceY } from "d3-force";
import { normalizeName } from "./lib/names.mjs";

export const MISC_CLUSTER = "misc";
export const MIN_CLUSTER_SIZE = 3;
export const PALETTE = ["#c8361f", "#3f6b3a", "#2f5d8a", "#e0a526", "#7a3f8f", "#d36b9e", "#4a9a9a", "#a8661c", "#5b7f2a", "#8a3b52", "#3d6f8f", "#b5892f"];
export const MISC_COLOR = "#8a8a7a";
const NAME_BANNED_TAGS = new Set(["alternative"]);

export function mulberry32(seed) {
  let state = seed >>> 0;
  return function next() {
    state = (state + 0x6d2b79f5) >>> 0;
    let value = state;
    value = Math.imul(value ^ (value >>> 15), value | 1);
    value ^= value + Math.imul(value ^ (value >>> 7), value | 61);
    return ((value ^ (value >>> 14)) >>> 0) / 4294967296;
  };
}

const edgeKey = (left, right) => (left < right ? `${left}|${right}` : `${right}|${left}`);
const tagSet = (artist) => new Set([...(artist.lfmTags || []).map((tag) => tag.name), ...(artist.mbGenres || [])]);

function jaccard(left, right) {
  if (!left.size || !right.size) return 0;
  let shared = 0;
  for (const tag of left) if (right.has(tag)) shared += 1;
  return shared / (left.size + right.size - shared);
}

export function buildEdges(artists) {
  const idByName = new Map(artists.map((artist) => [normalizeName(artist.name), artist.id]));
  const weights = new Map();
  const add = (left, right, weight) => {
    if (left === right) return;
    const key = edgeKey(left, right);
    weights.set(key, Math.max(weights.get(key) || 0, weight));
  };
  for (const artist of artists) {
    for (const similar of artist.lfmSimilar || []) {
      const otherId = idByName.get(normalizeName(similar.name));
      if (otherId) add(artist.id, otherId, similar.match);
    }
  }
  const connected = new Set([...weights.keys()].flatMap((key) => key.split("|")));
  const tags = new Map(artists.map((artist) => [artist.id, tagSet(artist)]));
  for (const artist of artists) {
    if (connected.has(artist.id)) continue;
    const others = artists.filter((other) => other.id !== artist.id);
    let fallback = others
      .map((other) => ({ id: other.id, weight: jaccard(tags.get(artist.id), tags.get(other.id)) * 0.5 }))
      .filter((candidate) => candidate.weight >= 0.1);
    const firstGenre = artist.appleGenres?.[0];
    if (fallback.length === 0 && firstGenre) {
      fallback = others.filter((other) => other.appleGenres?.[0] === firstGenre).map((other) => ({ id: other.id, weight: 0.1 }));
    }
    fallback
      .sort((left, right) => right.weight - left.weight || left.id.localeCompare(right.id))
      .slice(0, 3)
      .forEach((candidate) => add(artist.id, candidate.id, candidate.weight));
  }
  return [...weights.entries()]
    .map(([key, weight]) => {
      const [source, target] = key.split("|");
      return { source, target, weight: Math.round(weight * 1000) / 1000 };
    })
    .sort((left, right) => left.source.localeCompare(right.source) || left.target.localeCompare(right.target));
}

export function detectCommunities(ids, edges, { resolution = 1, seed = 42 } = {}) {
  const sortedIds = [...ids].sort();
  const graph = new Graph({ type: "undirected" });
  for (const id of sortedIds) graph.addNode(id);
  for (const edge of edges) {
    if (graph.hasNode(edge.source) && graph.hasNode(edge.target)) graph.mergeEdge(edge.source, edge.target, { weight: edge.weight });
  }
  const community = louvain(graph, { resolution, rng: mulberry32(seed), getEdgeWeight: "weight" });
  const members = new Map();
  for (const id of sortedIds) {
    if (!members.has(community[id])) members.set(community[id], []);
    members.get(community[id]).push(id);
  }
  const groups = [...members.values()].sort((left, right) => right.length - left.length || left[0].localeCompare(right[0]));
  const clusterOf = new Map();
  let index = 0;
  for (const group of groups) {
    const clusterId = group.length >= MIN_CLUSTER_SIZE ? `c${index++}` : MISC_CLUSTER;
    for (const id of group) clusterOf.set(id, clusterId);
  }
  return clusterOf;
}

const titleCase = (text) => text.replace(/\b\w/g, (letter) => letter.toUpperCase());

export function nameClusters(clusterOf, artists, nameOverrides = {}) {
  const byCluster = new Map();
  for (const artist of artists) {
    const clusterId = clusterOf.get(artist.id);
    if (!byCluster.has(clusterId)) byCluster.set(clusterId, []);
    byCluster.get(clusterId).push(artist);
  }
  const order = (id) => (id === MISC_CLUSTER ? Infinity : Number(id.slice(1)));
  const clusterIds = [...byCluster.keys()].sort((left, right) => order(left) - order(right));
  return clusterIds.map((id, index) => {
    const members = byCluster.get(id);
    const override = members.map((member) => nameOverrides[member.id]).find(Boolean);
    if (id === MISC_CLUSTER) return { id, name: override || "Everything else", color: MISC_COLOR, size: members.length };
    const scores = new Map();
    for (const member of members) {
      for (const tag of member.lfmTags || []) scores.set(tag.name, (scores.get(tag.name) || 0) + tag.count / 100);
      for (const genre of member.mbGenres || []) scores.set(genre, (scores.get(genre) || 0) + 0.5);
    }
    const topTags = [...scores.entries()]
      .filter(([tag]) => !NAME_BANNED_TAGS.has(tag))
      .sort((left, right) => right[1] - left[1] || left[0].localeCompare(right[0]))
      .slice(0, 2)
      .map(([tag]) => titleCase(tag));
    return { id, name: override || topTags.join(" / ") || "Unsorted", color: PALETTE[index % PALETTE.length], size: members.length };
  });
}

export function computePositions(ids, edges, previous = {}, { seed = 42, ticks = 400, anchorStrength = 0.5 } = {}) {
  const random = mulberry32(seed);
  const neighbours = new Map(ids.map((id) => [id, []]));
  for (const edge of edges) {
    neighbours.get(edge.source)?.push(edge.target);
    neighbours.get(edge.target)?.push(edge.source);
  }
  const nodes = [...ids].sort().map((id) => {
    const anchor = previous[id];
    if (anchor) return { id, x: anchor.x, y: anchor.y, anchor };
    const placedNeighbour = neighbours.get(id).map((neighbourId) => previous[neighbourId]).find(Boolean);
    const angle = random() * Math.PI * 2;
    if (placedNeighbour) return { id, x: placedNeighbour.x + Math.cos(angle) * 20, y: placedNeighbour.y + Math.sin(angle) * 20, anchor: null };
    const radius = 200 * Math.sqrt(random());
    return { id, x: Math.cos(angle) * radius, y: Math.sin(angle) * radius, anchor: null };
  });
  const known = new Set(ids);
  const links = edges.filter((edge) => known.has(edge.source) && known.has(edge.target)).map((edge) => ({ ...edge }));
  forceSimulation(nodes)
    .randomSource(random)
    .force("link", forceLink(links).id((node) => node.id).distance((link) => 30 + (1 - link.weight) * 90).strength((link) => 0.1 + link.weight * 0.6))
    .force("charge", forceManyBody().strength(-60))
    .force("collide", forceCollide(10))
    .force("anchorX", forceX((node) => (node.anchor ? node.anchor.x : 0)).strength((node) => (node.anchor ? anchorStrength : 0.01)))
    .force("anchorY", forceY((node) => (node.anchor ? node.anchor.y : 0)).strength((node) => (node.anchor ? anchorStrength : 0.01)))
    .stop()
    .tick(ticks);
  return Object.fromEntries(nodes.map((node) => [node.id, { x: Math.round(node.x * 10) / 10, y: Math.round(node.y * 10) / 10 }]));
}

export function runLayout(artists, previous, { clusterNames = {}, resolution = 1, seed = 42 } = {}) {
  const ids = artists.map((artist) => artist.id);
  const edges = buildEdges(artists);
  const clusterOf = detectCommunities(ids, edges, { resolution, seed });
  const clusters = nameClusters(clusterOf, artists, clusterNames);
  const positions = computePositions(ids, edges, previous, { seed });
  return { edges, clusterOf, clusters, positions };
}

export function assembleData({ artists, edges, clusterOf, clusters, positions, stories, stats, generated }) {
  const idByMbid = new Map(artists.filter((artist) => artist.mbid).map((artist) => [artist.mbid, artist.id]));
  const similarById = new Map(artists.map((artist) => [artist.id, []]));
  for (const edge of edges) {
    similarById.get(edge.source)?.push({ id: edge.target, weight: edge.weight });
    similarById.get(edge.target)?.push({ id: edge.source, weight: edge.weight });
  }
  return {
    generated,
    stats,
    clusters: clusters.map(({ id, name, color }) => ({ id, name, color })),
    artists: artists.map((artist) => ({
      id: artist.id,
      name: artist.name,
      mbid: artist.mbid,
      cluster: clusterOf.get(artist.id),
      x: positions[artist.id].x,
      y: positions[artist.id].y,
      love: artist.love,
      songs: artist.songs,
      plays: artist.plays,
      favourites: artist.favourites,
      appleGenres: artist.appleGenres,
      tracks: artist.tracks,
      similar: similarById.get(artist.id).sort((left, right) => right.weight - left.weight || left.id.localeCompare(right.id)).slice(0, 8),
      lineage: artist.lineage.map((item) => ({ ...item, id: idByMbid.get(item.mbid) || null })),
    })),
    stories,
  };
}

export function validateData(data, { forbiddenStrings = [] } = {}) {
  const errors = [];
  const ids = new Set();
  const clusterIds = new Set(data.clusters.map((cluster) => cluster.id));
  for (const artist of data.artists) {
    if (ids.has(artist.id)) errors.push(`duplicate artist id: ${artist.id}`);
    ids.add(artist.id);
    if (!clusterIds.has(artist.cluster)) errors.push(`${artist.id}: unknown cluster ${artist.cluster}`);
    if (!Number.isFinite(artist.x) || !Number.isFinite(artist.y)) errors.push(`${artist.id}: position is not a finite number`);
  }
  for (const artist of data.artists) {
    for (const similar of artist.similar) if (!ids.has(similar.id)) errors.push(`${artist.id}: similar artist ${similar.id} does not exist`);
    for (const item of artist.lineage) if (item.id && !ids.has(item.id)) errors.push(`${artist.id}: lineage id ${item.id} does not exist`);
  }
  for (const story of data.stories) {
    for (const step of story.steps) if (!ids.has(step.artist)) errors.push(`story ${story.id}: artist ${step.artist} is not on the map`);
  }
  const text = JSON.stringify(data);
  for (const secret of forbiddenStrings) {
    if (secret && text.includes(secret)) errors.push("output contains a forbidden string (an API key?)");
  }
  return errors;
}

export function largestMoves(previous, positions, limit = 5) {
  return Object.keys(positions)
    .filter((id) => previous[id])
    .map((id) => ({ id, distance: Math.round(Math.hypot(positions[id].x - previous[id].x, positions[id].y - previous[id].y) * 10) / 10 }))
    .filter((move) => move.distance > 0)
    .sort((left, right) => right.distance - left.distance || left.id.localeCompare(right.id))
    .slice(0, limit);
}

export async function loadStories(dir) {
  let files;
  try {
    files = (await readdir(dir)).filter((file) => file.endsWith(".json")).sort();

    throw error;
  }

}

export const MIN_SCALE = 0.2;
export const MAX_SCALE = 6;

export function createCamera(width, height) {
  return { x: 0, y: 0, scale: 1, width, height };
}

export function worldToScreen(camera, worldX, worldY) {
  return [(worldX - camera.x) * camera.scale + camera.width / 2, (worldY - camera.y) * camera.scale + camera.height / 2];
}

export function screenToWorld(camera, screenX, screenY) {
  return [(screenX - camera.width / 2) / camera.scale + camera.x, (screenY - camera.height / 2) / camera.scale + camera.y];
}

export function clampScale(scale) {
  return Math.min(MAX_SCALE, Math.max(MIN_SCALE, scale));
}

export function zoomAt(camera, screenX, screenY, factor) {
  const [worldX, worldY] = screenToWorld(camera, screenX, screenY);
  const zoomed = { ...camera, scale: clampScale(camera.scale * factor) };
  const [afterX, afterY] = screenToWorld(zoomed, screenX, screenY);
  return { ...zoomed, x: camera.x + (worldX - afterX), y: camera.y + (worldY - afterY) };
}

export function panBy(camera, deltaX, deltaY) {
  return { ...camera, x: camera.x - deltaX / camera.scale, y: camera.y - deltaY / camera.scale };
}

export function fitBounds(camera, points, padding = 40) {
  if (points.length === 0) return camera;
  const xs = points.map((point) => point.x);
  const ys = points.map((point) => point.y);
  const [minX, maxX, minY, maxY] = [Math.min(...xs), Math.max(...xs), Math.min(...ys), Math.max(...ys)];
  const scale = clampScale(Math.min((camera.width - padding * 2) / (maxX - minX || 1), (camera.height - padding * 2) / (maxY - minY || 1)));
  return { ...camera, x: (minX + maxX) / 2, y: (minY + maxY) / 2, scale };
}

export function nodeRadius(love) {
  return 2 + 1.6 * Math.sqrt(Math.max(0, love));
}

export function hitTest(nodes, worldX, worldY, slop) {
  let best = null;
  let bestDistance = Infinity;
  for (const node of nodes) {
    const distance = Math.hypot(node.x - worldX, node.y - worldY);
    if (distance <= node.r + slop && distance < bestDistance) {
      best = node;
      bestDistance = distance;
    }
  }
  return best;
}

export function labelVisible(radius, scale, emphasised) {
  return emphasised || radius * scale >= 9;
}

const SWELL = 1.35;
const PULSE_MS = 600;

export function stepSpring(spring, target, dt, stiffness = 170, damping = 14) {
  const force = -stiffness * (spring.value - target) - damping * spring.velocity;
  const velocity = spring.velocity + force * dt;
  return { value: spring.value + velocity * dt, velocity };
}

export function isSettled(spring, target, epsilon = 0.01) {
  return Math.abs(spring.value - target) < epsilon && Math.abs(spring.velocity) < epsilon;
}

export function easeInOutCubic(t) {
  return t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2;
}

export function lerpCamera(from, to, t) {
  if (t >= 1) return { ...to };
  const eased = easeInOutCubic(t);
  return {
    ...from,
    x: from.x + (to.x - from.x) * eased,
    y: from.y + (to.y - from.y) * eased,
    scale: from.scale * (to.scale / from.scale) ** eased,
  };
}

export function createJuice({ reducedMotion = false } = {}) {
  const offsets = new Map();
  const swells = new Map();
  const pulses = [];
  let hoveredId = null;
  const swellTarget = (id) => (id === hoveredId ? SWELL : 1);

  return {
    reducedMotion,
    hover(id) {
      if (id === hoveredId) return;
      hoveredId = id;
      if (id && !swells.has(id)) swells.set(id, { value: 1, velocity: 0 });
    },
    poke(node, neighbours) {
      if (reducedMotion) return;
      for (const { node: other, weight } of neighbours) {
        const dx = other.x - node.x;
        const dy = other.y - node.y;
        const distance = Math.hypot(dx, dy) || 1;
        const kick = 260 * (0.3 + weight);
        const offset = offsets.get(other.id) || { x: { value: 0, velocity: 0 }, y: { value: 0, velocity: 0 } };
        offset.x.velocity += (dx / distance) * kick;
        offset.y.velocity += (dy / distance) * kick;
        offsets.set(other.id, offset);
      }
    },
    pulse(x, y, now) {
      if (!reducedMotion) pulses.push({ x, y, start: now });
    },
    offset(id) {
      const offset = offsets.get(id);
      return offset ? [offset.x.value, offset.y.value] : [0, 0];
    },
    scale(id) {
      if (reducedMotion) return swellTarget(id);
      return swells.get(id)?.value ?? 1;
    },
    activePulses(now) {
      return pulses.filter((pulse) => now - pulse.start < PULSE_MS).map((pulse) => ({ x: pulse.x, y: pulse.y, progress: (now - pulse.start) / PULSE_MS }));
    },
    step(dt, now) {
      if (reducedMotion) return false;
      let moving = false;
      for (const [id, offset] of offsets) {
        offset.x = stepSpring(offset.x, 0, dt);
        offset.y = stepSpring(offset.y, 0, dt);
        if (isSettled(offset.x, 0) && isSettled(offset.y, 0)) offsets.delete(id);
        else moving = true;
      }
      for (const [id, spring] of swells) {
        const target = swellTarget(id);
        const next = stepSpring(spring, target, dt, 300, 18);
        if (!isSettled(next, target)) {
          swells.set(id, next);
          moving = true;
        } else if (target === 1) swells.delete(id);
        else swells.set(id, { value: target, velocity: 0 });
      }
      while (pulses.length && now - pulses[0].start >= PULSE_MS) pulses.shift();
      return moving || pulses.length > 0;
    },
  };
}

}

export function parseHash(hash) {
  const params = new URLSearchParams(String(hash).replace(/^#/, ""));
  const step = Number.parseInt(params.get("step") || "0", 10);
  return { artist: params.get("artist") || null, story: params.get("story") || null, step: Number.isFinite(step) && step >= 0 ? step : 0 };
}

export function formatHash({ artist, story, step } = {}) {
  const params = new URLSearchParams();
  if (story) {
    params.set("story", story);
    if (step) params.set("step", String(step));
  }
  if (artist) params.set("artist", artist);
  const text = params.toString();
  return text ? `#${text}` : "";
}

export function findArtists(query, artists, limit = 8) {
  const wanted = normalise(query);
  if (!wanted) return [];
  return artists
    .map((artist) => ({ artist, name: normalise(artist.name) }))
    .filter((entry) => entry.name.includes(wanted))
    .sort((left, right) =>
      Number(right.name.startsWith(wanted)) - Number(left.name.startsWith(wanted)) ||
      right.artist.love - left.artist.love ||
      left.artist.name.localeCompare(right.artist.name))
    .slice(0, limit)
    .map((entry) => entry.artist);
}

export function wireSearch(input, datalist, artists) {
  for (const artist of [...artists].sort((left, right) => left.name.localeCompare(right.name))) {
    const option = document.createElement("option");
    option.value = artist.name;
    datalist.append(option);
  }
  input.addEventListener("change", () => {
    const [match] = findArtists(input.value, artists, 1);
    if (!match) return;
    location.hash = formatHash({ artist: match.id });
    input.value = "";
    input.blur();
  });
}

export function lineageLayout(count, radius = 90) {
  return Array.from({ length: count }, (_, index) => {

    else node.setAttribute(key, value);
  }

  return node;
}

import { createCamera, worldToScreen, screenToWorld, zoomAt, panBy, fitBounds, nodeRadius, hitTest, labelVisible } from "./camera.js";
import { lerpCamera } from "./juice.js";

const CLICK_SLOP_PX = 5;
const GLIDE_MS = 650;

function readTheme() {
  const style = getComputedStyle(document.documentElement);
  const token = (name) => style.getPropertyValue(name).trim();
  return { bg: token("--bg"), ink: token("--ink"), muted: token("--muted"), accent: token("--stamp") };
}

export function createMap({ canvas, data, juice, onSelect }) {
  const context = canvas.getContext("2d");
  const clusterColor = new Map(data.clusters.map((cluster) => [cluster.id, cluster.color]));
  const nodes = data.artists.map((artist) => ({ ...artist, r: nodeRadius(artist.love) }));
  const byId = new Map(nodes.map((node) => [node.id, node]));
  let camera = createCamera(1, 1);
  let theme = readTheme();
  let tween = null;
  let hoveredId = null;
  let selectedId = null;
  let story = null;
  let frameRequested = false;
  let lastFrame = 0;

  const neighboursOf = (node) => node.similar.map((link) => ({ node: byId.get(link.id), weight: link.weight })).filter((link) => link.node);
  const position = (node) => { const [dx, dy] = juice.offset(node.id); return [node.x + dx, node.y + dy]; };
  const screenOf = (node) => worldToScreen(camera, ...position(node));

  function requestFrame() {
    if (frameRequested) return;
    frameRequested = true;
    requestAnimationFrame(frame);
  }

  function frame(now) {
    frameRequested = false;
    const dt = Math.min(1 / 30, lastFrame ? (now - lastFrame) / 1000 : 1 / 60);
    lastFrame = now;
    let animating = juice.step(dt, now);
    if (tween) {
      const progress = Math.min(1, (now - tween.start) / GLIDE_MS);
      camera = { ...lerpCamera(tween.from, tween.to, progress), width: camera.width, height: camera.height };
      if (progress < 1) animating = true;
      else tween = null;
    }
    draw(now);
    if (animating) requestFrame();
    else lastFrame = 0;
  }

  function drawLines(focusId) {
    const focus = byId.get(focusId);
    if (!focus) return;
    const [fx, fy] = screenOf(focus);
    context.strokeStyle = theme.muted;
    for (const { node, weight } of neighboursOf(focus)) {
      const [nx, ny] = screenOf(node);
      context.globalAlpha = 0.25 + weight * 0.6;
      context.lineWidth = 1 + weight * 2;
      context.beginPath();
      context.moveTo(fx, fy);
      context.lineTo(nx, ny);
      context.stroke();
    }
    context.globalAlpha = 1;
  }

  function drawStoryPath() {
    const points = story.ids.slice(0, story.current + 1).map((id) => byId.get(id)).filter(Boolean).map(screenOf);
    if (points.length < 2) return;
    context.strokeStyle = theme.accent;
    context.lineWidth = 3;
    context.setLineDash([8, 6]);
    context.beginPath();
    points.forEach(([x, y], index) => (index === 0 ? context.moveTo(x, y) : context.lineTo(x, y)));
    context.stroke();
    context.setLineDash([]);
  }

  function draw(now) {
    const { width, height } = camera;
    const focusId = hoveredId || selectedId;
    const storyIds = story ? new Set(story.ids) : null;
    context.fillStyle = theme.bg;
    context.fillRect(0, 0, width, height);
    if (focusId) drawLines(focusId);
    if (story) drawStoryPath();
    for (const node of nodes) {
      const [x, y] = screenOf(node);
      const radius = Math.max(1.5, node.r * camera.scale * juice.scale(node.id));
      if (x < -radius || y < -radius || x > width + radius || y > height + radius) continue;
      context.globalAlpha = storyIds && !storyIds.has(node.id) ? 0.15 : 1;
      context.fillStyle = clusterColor.get(node.cluster) || theme.muted;
      context.beginPath();
      context.arc(x, y, radius, 0, Math.PI * 2);
      context.fill();
      if (node.id === selectedId) {
        context.strokeStyle = theme.ink;
        context.lineWidth = 2;
        context.stroke();
      }
    }
    context.globalAlpha = 1;
    for (const pulse of juice.activePulses(now)) {
      const [x, y] = worldToScreen(camera, pulse.x, pulse.y);
      context.globalAlpha = 1 - pulse.progress;
      context.strokeStyle = theme.accent;
      context.lineWidth = 3;
      context.beginPath();
      context.arc(x, y, 10 + pulse.progress * 50, 0, Math.PI * 2);
      context.stroke();
    }
    context.globalAlpha = 1;
    context.font = "12px 'Space Mono', monospace";
    context.textAlign = "center";
    context.textBaseline = "top";
    for (const node of nodes) {
      const inStory = storyIds?.has(node.id);
      if (storyIds && !inStory) continue;
      if (!labelVisible(node.r, camera.scale, node.id === focusId || inStory)) continue;
      const [x, y] = screenOf(node);
      const labelY = y + node.r * camera.scale * juice.scale(node.id) + 3;
      context.lineWidth = 3;
      context.strokeStyle = theme.bg;
      context.strokeText(node.name, x, labelY);
      context.fillStyle = theme.ink;
      context.fillText(node.name, x, labelY);
    }
  }

  function flyTo(id) {
    const node = byId.get(id);
    if (!node) return;
    const target = { ...camera, x: node.x, y: node.y, scale: Math.max(camera.scale, 1.6) };
    if (juice.reducedMotion) camera = target;
    else tween = { from: { ...camera }, to: target, start: performance.now() };
    requestFrame();
  }

  function select(id) {
    selectedId = byId.has(id) ? id : null;
    const node = byId.get(selectedId);
    if (node) {
      juice.poke(node, neighboursOf(node));
      juice.pulse(node.x, node.y, performance.now());
    }
    requestFrame();
  }

  function setStory(next) {
    story = next;
    requestFrame();
  }

  function resize() {
    const ratio = window.devicePixelRatio || 1;
    const { width, height } = canvas.getBoundingClientRect();
    canvas.width = Math.round(width * ratio);
    canvas.height = Math.round(height * ratio);
    context.setTransform(ratio, 0, 0, ratio, 0, 0);
    camera = { ...camera, width, height };
    requestFrame();
  }

  const pointers = new Map();
  let drag = null;
  let pinchDistance = null;
  const local = (event) => { const rect = canvas.getBoundingClientRect(); return [event.clientX - rect.left, event.clientY - rect.top]; };
  const nodeAt = (event) => hitTest(nodes, ...screenToWorld(camera, ...local(event)), 4 / camera.scale);

  canvas.addEventListener("pointerdown", (event) => {
    canvas.setPointerCapture(event.pointerId);
    pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
    drag = { x: event.clientX, y: event.clientY, moved: false };
    tween = null;
  });
  canvas.addEventListener("pointermove", (event) => {
    const previous = pointers.get(event.pointerId);
    if (!previous) {
      const hit = nodeAt(event);
      if ((hit?.id ?? null) === hoveredId) return;
      hoveredId = hit?.id ?? null;
      juice.hover(hoveredId);
      if (hit) juice.poke(hit, neighboursOf(hit));
      canvas.style.cursor = hit ? "pointer" : "grab";
      requestFrame();
      return;
    }
    pointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
    if (pointers.size === 2) {
      const [first, second] = [...pointers.values()];
      const distance = Math.hypot(first.x - second.x, first.y - second.y);
      const rect = canvas.getBoundingClientRect();
      if (pinchDistance) camera = zoomAt(camera, (first.x + second.x) / 2 - rect.left, (first.y + second.y) / 2 - rect.top, distance / pinchDistance);
      pinchDistance = distance;
      drag.moved = true;
    } else {
      camera = panBy(camera, event.clientX - previous.x, event.clientY - previous.y);
      if (Math.hypot(event.clientX - drag.x, event.clientY - drag.y) > CLICK_SLOP_PX) drag.moved = true;
    }
    requestFrame();
  });
  function endPointer(event) {
    pointers.delete(event.pointerId);
    if (pointers.size < 2) pinchDistance = null;
    if (pointers.size === 0 && drag && !drag.moved && event.type === "pointerup") onSelect(nodeAt(event)?.id ?? null);
    if (pointers.size === 0) drag = null;
  }
  canvas.addEventListener("pointerup", endPointer);
  canvas.addEventListener("pointercancel", endPointer);
  canvas.addEventListener("pointerleave", () => {
    if (!hoveredId || pointers.size) return;
    hoveredId = null;
    juice.hover(null);
    requestFrame();
  });
  canvas.addEventListener("wheel", (event) => {
    event.preventDefault();
    camera = zoomAt(camera, ...local(event), Math.exp(-event.deltaY * 0.0015));
    tween = null;
    requestFrame();
  }, { passive: false });
  canvas.addEventListener("keydown", (event) => {
    const pans = { ArrowLeft: [60, 0], ArrowRight: [-60, 0], ArrowUp: [0, 60], ArrowDown: [0, -60] };
    const zooms = { "+": 1.25, "=": 1.25, "-": 0.8 };
    if (pans[event.key]) camera = panBy(camera, ...pans[event.key]);
    else if (zooms[event.key]) camera = zoomAt(camera, camera.width / 2, camera.height / 2, zooms[event.key]);
    else return;
    event.preventDefault();
    requestFrame();
  });
  window.addEventListener("resize", resize);
  window.matchMedia("(prefers-color-scheme: dark)").addEventListener("change", () => { theme = readTheme(); requestFrame(); });

  resize();
  camera = fitBounds(camera, nodes);
  requestFrame();
  return { flyTo, select, setStory, byId };
}

`static/music/stories.js`:
```js
import { formatHash } from "./search.js";

export function clampStep(step, story) {
  return Math.min(Math.max(0, step), story.steps.length - 1);
}

function textElement(tag, className, text) {
  const node = document.createElement(tag);
  if (className) node.className = className;
  node.textContent = text;
  return node;
}

function navButton(label, hash) {
  const button = textElement("button", "btn-plain", label);
  button.type = "button";
  if (hash === null) button.disabled = true;
  else button.addEventListener("click", () => { location.hash = hash; });
  return button;
}

export function createStories({ select, box, data, map }) {
  const storiesById = new Map(data.stories.map((story) => [story.id, story]));
  for (const story of data.stories) {
    const option = textElement("option", "", story.title);
    option.value = story.id;
    select.append(option);
  }
  select.addEventListener("change", () => { location.hash = select.value ? formatHash({ story: select.value }) : ""; });
  let openStoryId = null;

  function render(story, step) {
    const current = story.steps[step];
    const isLast = step === story.steps.length - 1;
    const controls = document.createElement("div");
    controls.className = "story-controls";
    controls.append(
      navButton("Back", step > 0 ? formatHash({ story: story.id, step: step - 1 }) : null),
      navButton(isLast ? "Done" : "Next", isLast ? "#" : formatHash({ story: story.id, step: step + 1 })),
    );
    if (story.link) {
      const link = textElement("a", "story-link", story.link.label);
      link.href = story.link.href;
      controls.append(link);
    }
    box.replaceChildren(
      textElement("p", "kicker", `${story.title} · ${step + 1} of ${story.steps.length}`),
      textElement("h2", "", map.byId.get(current.artist)?.name ?? current.artist),
      textElement("p", "", current.caption),
      controls,
    );
    box.hidden = false;
  }

  return {
    open(id, step) {
      const story = storiesById.get(id);
      if (!story) return false;
      const safeStep = clampStep(step, story);
      openStoryId = id;
      select.value = id;
      map.setStory({ ids: story.steps.map((item) => item.artist), current: safeStep });
      map.flyTo(story.steps[safeStep].artist);
      render(story, safeStep);
      return true;
    },
    close() {
      if (!openStoryId) return;
      openStoryId = null;
      select.value = "";
      map.setStory(null);
      box.hidden = true;
      box.replaceChildren();
    },
  };
}
```
